```css
/* ── Layout Shell: portrait-first responsive grid ──────────── */

.layout-shell {

  --safe-bottom: env(safe-area-inset-bottom, 0px);
  --safe-left: env(safe-area-inset-left, 0px);
  --safe-right: env(safe-area-inset-right, 0px);
  padding-top: var(--safe-top);
  padding-left: var(--safe-left);
  padding-right: var(--safe-right);
}

.layout-shell__canvas {
  grid-area: canvas;
  overflow: hidden;
  min-height: 0;
}

.layout-shell__hud {
  grid-area: hud;
  min-height: 0;
}

.layout-shell__interaction {
  grid-area: interaction;
  overflow: hidden;
  min-height: 0;
  padding-bottom: var(--safe-bottom);
}

/* Tablet (768px - 1023px): slightly less canvas per D-05 */
@media (min-width: 768px) and (max-width: 1023px) {
  .layout-shell {
    grid-template-rows: clamp(55%, 60vh, 70%) 60px 1fr;
  }
}

/* Desktop (>= 1024px): full-canvas overlay mode per D-03 */
@media (min-width: 1024px) {
  .layout-shell {
    grid-template-rows: 1fr;
    grid-template-areas: "canvas";
  }

  /* HUD and interaction become floating overlays on desktop */
  .layout-shell__hud,
  .layout-shell__interaction {
    position: absolute;
    pointer-events: none;
  }

  .layout-shell__hud > *,
  .layout-shell__interaction > * {
    pointer-events: auto;
  }
}
```
